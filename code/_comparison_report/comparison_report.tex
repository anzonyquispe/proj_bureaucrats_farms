% \safefig{<basename>}{<width>}{<caption hint>}
%   Includes <basename>.png from \figdir; shows (pending) if the file is

\newcommand{\safefig}[3]{%
  \IfFileExists{\figdir/#1.png}{%
    \includegraphics[width=#2]{#1.png}%
  }{%
    \fbox{\parbox[c][0.45\linewidth][c]{0.95\linewidth}{%

    }}%
  }%

% ---------------------------------------------------------------------------
% \safeTab{<stem>}
%   \inputs <stem>.tex from \tablesdir, scaled to fit \linewidth if wider.
%   Shows (pending) if the file is missing.
% ---------------------------------------------------------------------------
\newcommand{\safeTab}[1]{%
  \IfFileExists{\tablesdir/#1.tex}{%
    \begin{adjustbox}{max width=\linewidth, center}%
      \input{\tablesdir/#1.tex}%
    \end{adjustbox}%
  }{%
    \smallskip
    \fbox{\parbox{\linewidth}{\centering\color{red}\itshape(pending: #1.tex)}}%
    \smallskip
  }%
}

\noindent
This document compares all rural results under the original treatment
\texttt{downup\_ac} and the population-weighted variant \texttt{downup\_ac\_pop},
across the full rural-replication, stacked event-study, and dCDH pipelines.

\medskip\noindent
$^\dagger$\,\textbf{HonestDiD panels (Sections~2--6):} each treatment row
shows a 2$\times$2 grid.  Top row: original event-study coefficients
(\texttt{\_ori}) and their rotation-based version (\texttt{\_rotated}).
Bottom row: \emph{relative-magnitudes} sensitivity bounds from
\texttt{HonestDiD::createSensitivityResults\_relativeMagnitudes} applied to
the \textbf{original} coefficients (\texttt{\_honest1}) and to the
\textbf{rotated} coefficients (\texttt{\_rot\_honest1}).  Other files on disk
but not shown: \texttt{\_honest2} and \texttt{\_rot\_honest2}
(smoothness-restriction bound on original and rotated, respectively).

\medskip\noindent
Missing figures or tables are flagged inline as
{\color{red}\itshape(pending: \dots)} and do not break compilation.
